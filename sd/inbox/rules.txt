# Pocket Inbox rules. Copy this folder to the SD card as /inbox/.
#
# One question per line:   <type> <key>: <question text> | <option>, <option>, ...
#   type  choice | score | noul | span   (score options go low -> high; noul and span take no options)
#   key   field name used in the saved files (letters, digits, _)
#
# [route]  exactly one choice question; its options are the categories.
# [all]    asked for every entry, in the same pass as [route].
# [<cat>]  asked only when the entry was routed to <cat> (second pass).
#          <cat> = the option lowercased, spaces -> _ ("shopping list" -> [shopping_list]).
#
# A new category = a new option on the route line (+ an optional [<cat>] section).
# Editing a question's wording resets the corrections learned for it.

[route]
choice route: What kind of entry is this? | todo, event, expense, contact, note, idea, shopping

[all]
score urgency: How urgent is it? | low, medium, high
span time: Extract the time.
span day: Extract the day or date.
span person: Extract the person's name.

[expense]
span amount: Extract the amount of money.
choice spent_on: What was the money spent on? | food, transport, shopping, bills, fun, other

[contact]
span phone: Extract the phone number.
span email: Extract the email address.

[event]
span place: Extract the place.
